cmake_minimum_required(VERSION 3.21)
project(yw1nx_foundation VERSION 0.1.0 LANGUAGES CXX)

add_library(yw1nx_foundation STATIC
    source/game/versions/version_manager.cpp
    source/game/symbols/symbol_registry.cpp
)
target_include_directories(yw1nx_foundation PUBLIC source)
target_compile_features(yw1nx_foundation PUBLIC cxx_std_20)
if(MSVC)
    target_compile_options(yw1nx_foundation PRIVATE /W4 /WX /utf-8)
else()
    target_compile_options(yw1nx_foundation PRIVATE
        -Wall -Wextra -Wpedantic -Werror -fno-exceptions -fno-rtti)
endif()

option(YW1_BUILD_BOOTSTRAP "Build the Stage B.0.1 bootstrap module" OFF)
option(YW1_BOOTSTRAP_SD_LOGGING "Enable the bounded independent SD logging worker" OFF)
option(YW1_BOOTSTRAP_FS_DIAGNOSTIC "Isolate FS diagnostics from exlaunch initialization" OFF)
option(YW1_BOOTSTRAP_EXTERNAL_DIAGNOSTICS "Expose the bounded B.0.1.5 diagnostic ABI" OFF)
option(YW1_BUILD_DIAGNOSTICS_COMPANION "Build the read-only diagnostics NRO" OFF)
option(YW1_BOOTSTRAP_EXLAUNCH_DIAGNOSTIC "Verify exlaunch initialization without a direct FS writer" OFF)
option(YW1_BOOTSTRAP_EGL_DIAGNOSTIC "Inspect EGL hook prerequisites without installing a hook" OFF)
option(YW1_BOOTSTRAP_SAFE_HOOK_DIAGNOSTIC "Report blocked safe-hook prerequisites without installing" OFF)
option(YW1_COMPANION_EGL_GOT_PROBE "Build a read-only GOT companion for the unchanged B021 module" OFF)
option(YW1_COMPANION_EGL_TARGET_MEMORY "Query only the captured GOT target in the game process" OFF)
option(YW1_COMPANION_EGL_CODE_SNAPSHOT "Read at most 16 bytes from a validated RX GOT target" OFF)
option(YW1_COMPANION_EGL_CODE_ATTRIBUTION "Read a fixed 64-byte GOT target snapshot for host analysis" OFF)
option(YW1_COMPANION_SALTY_ORIGINAL_EGL "Inspect the fixed SaltyNX reference original EGL slot" OFF)
set(YW1_EGL_CALLBACK_PROFILE "OFF" CACHE STRING "OFF, READ_ONLY or CANARY")
set_property(CACHE YW1_EGL_CALLBACK_PROFILE PROPERTY STRINGS OFF READ_ONLY CANARY)
option(YW1_EGL_CONTEXT_PROBE "Enable the bounded B029 context publication and probe" OFF)
option(YW1_EGL_GETTER_WHITELIST "Enable the B0210 build-bound two-getter whitelist" OFF)
option(YW1_IMGUI_DEMO_DIAGNOSTICS "Publish the B0211 drawing-stop diagnostics" OFF)
option(YW1_IMGUI_DEMO_EXPERIMENTAL "Request experimental Profile C (currently drawing blocked)" OFF)
option(YW1_BUILD_IMGUI_REFERENCE "Compile pinned ImGui separately without module linkage" OFF)
option(YW1_BUILD_RENDER_FOUNDATION "Build allocator, resolver and protected official renderer" OFF)
option(YW1_RENDER_DIAGNOSTICS "Publish the B0212 live foundation diagnostics" OFF)
option(YW1_RENDER_MEMORY_DIAGNOSTICS "Publish B0213 Native memory obligations (drawing OFF)" OFF)
if(YW1_RENDER_MEMORY_DIAGNOSTICS AND NOT YW1_RENDER_DIAGNOSTICS)
    message(FATAL_ERROR "B0213 memory diagnostics require renderer diagnostics")
endif()
set(YW1_IMGUI_RENDER_MODE "OFF" CACHE STRING "OFF, MINIMAL or DEMO (native OOM gate remains closed)")
set_property(CACHE YW1_IMGUI_RENDER_MODE PROPERTY STRINGS OFF MINIMAL DEMO)
set(YW1_IMGUI_ARENA_BYTES "1048576" CACHE STRING "Module-owned ImGui arena limit")
if(NOT YW1_IMGUI_RENDER_MODE MATCHES "^(OFF|MINIMAL|DEMO)$")
    message(FATAL_ERROR "Unknown ImGui render mode")
endif()
if(YW1_RENDER_DIAGNOSTICS AND NOT YW1_EGL_GETTER_WHITELIST)
    message(FATAL_ERROR "Render diagnostics require the build-bound getter profile")
endif()
if(NOT YW1_IMGUI_RENDER_MODE STREQUAL "OFF" AND
   (NOT YW1_RENDER_DIAGNOSTICS OR NOT YW1_BUILD_RENDER_FOUNDATION OR
    NOT YW1_EGL_CALLBACK_PROFILE STREQUAL "CANARY" OR YW1_IMGUI_DEMO_DIAGNOSTICS))
    message(FATAL_ERROR "Render experiment requires foundation/CANARY and separate runtime ABI")
endif()
if(NOT YW1_IMGUI_ARENA_BYTES MATCHES "^[0-9]+$")
    message(FATAL_ERROR "Arena capacity must be an integer")
endif()
math(EXPR yw1_arena_remainder "${YW1_IMGUI_ARENA_BYTES} % 4096")
if(YW1_IMGUI_ARENA_BYTES LESS 65536 OR YW1_IMGUI_ARENA_BYTES GREATER 4194304 OR
   NOT yw1_arena_remainder EQUAL 0)
    message(FATAL_ERROR "Arena capacity must be page aligned and between 64KiB and 4MiB")
endif()
if(YW1_IMGUI_DEMO_DIAGNOSTICS AND NOT YW1_EGL_GETTER_WHITELIST)
    message(FATAL_ERROR "Demo diagnostics require the build-bound getter profile")
endif()
if(YW1_IMGUI_DEMO_EXPERIMENTAL AND
   (NOT YW1_IMGUI_DEMO_DIAGNOSTICS OR NOT YW1_EGL_CALLBACK_PROFILE STREQUAL "CANARY"))
    message(FATAL_ERROR "Experimental demo requires diagnostics and the CANARY profile")
endif()
if(YW1_EGL_GETTER_WHITELIST AND NOT YW1_EGL_CONTEXT_PROBE)
    message(FATAL_ERROR "Getter whitelist requires the context probe")
endif()
if(YW1_EGL_CONTEXT_PROBE AND YW1_EGL_CALLBACK_PROFILE STREQUAL "OFF")
    message(FATAL_ERROR "Context diagnostics require a callback profile")
endif()
if(NOT YW1_EGL_CALLBACK_PROFILE MATCHES "^(OFF|READ_ONLY|CANARY)$")
    message(FATAL_ERROR "Unknown EGL callback profile")
endif()
if(NOT YW1_EGL_CALLBACK_PROFILE STREQUAL "OFF" AND
   (NOT YW1_BUILD_BOOTSTRAP OR NOT YW1_BUILD_DIAGNOSTICS_COMPANION OR
    NOT YW1_COMPANION_SALTY_ORIGINAL_EGL OR NOT YW1_BOOTSTRAP_SAFE_HOOK_DIAGNOSTIC OR
    YW1_BOOTSTRAP_SD_LOGGING OR YW1_BOOTSTRAP_FS_DIAGNOSTIC))
    message(FATAL_ERROR "Callback profiles require the new bootstrap and complete B027 companion")
endif()
if(YW1_COMPANION_SALTY_ORIGINAL_EGL AND NOT YW1_COMPANION_EGL_CODE_ATTRIBUTION)
    message(FATAL_ERROR "Original EGL diagnostics require the 64-byte attribution profile")
endif()
if(YW1_COMPANION_EGL_CODE_ATTRIBUTION AND NOT YW1_COMPANION_EGL_CODE_SNAPSHOT)
    message(FATAL_ERROR "Code attribution requires the guarded code snapshot profile")
endif()
if(YW1_COMPANION_EGL_CODE_SNAPSHOT AND NOT YW1_COMPANION_EGL_TARGET_MEMORY)
    message(FATAL_ERROR "Code snapshots require the read-only target memory profile")
endif()
if(YW1_COMPANION_EGL_TARGET_MEMORY AND NOT YW1_COMPANION_EGL_GOT_PROBE)
    message(FATAL_ERROR "Target memory diagnostics require the read-only GOT companion")
endif()
option(YW1_EGL_HOOK_ENABLED "Enable the legacy EGL export trampoline after verification" OFF)
if(YW1_COMPANION_EGL_GOT_PROBE AND
    ((YW1_BUILD_BOOTSTRAP AND YW1_EGL_CALLBACK_PROFILE STREQUAL "OFF") OR
     NOT YW1_BUILD_DIAGNOSTICS_COMPANION OR
     NOT YW1_BOOTSTRAP_EXTERNAL_DIAGNOSTICS OR NOT YW1_BOOTSTRAP_EGL_DIAGNOSTIC OR
     NOT YW1_BOOTSTRAP_SAFE_HOOK_DIAGNOSTIC))
    message(FATAL_ERROR "GOT probe requires companion-only mode and the B021 external/EGL/safe ABI")
endif()
if(YW1_BOOTSTRAP_SAFE_HOOK_DIAGNOSTIC AND NOT YW1_BOOTSTRAP_EGL_DIAGNOSTIC)
    message(FATAL_ERROR "Safe hook diagnostics require the EGL diagnostic profile")
endif()
if(YW1_EGL_HOOK_ENABLED)
    message(FATAL_ERROR "EGL hook installer safety is unverified; hook-enabled artifacts are blocked")
endif()
if(YW1_BOOTSTRAP_EGL_DIAGNOSTIC AND
    (YW1_BOOTSTRAP_EXLAUNCH_DIAGNOSTIC OR
     (NOT YW1_BUILD_BOOTSTRAP AND NOT YW1_COMPANION_EGL_GOT_PROBE) OR
     NOT YW1_BOOTSTRAP_EXTERNAL_DIAGNOSTICS OR YW1_BOOTSTRAP_SD_LOGGING OR YW1_BOOTSTRAP_FS_DIAGNOSTIC))
    message(FATAL_ERROR "EGL diagnostics require bootstrap/external diagnostics and disable other profiles")
endif()
if(YW1_BOOTSTRAP_EXLAUNCH_DIAGNOSTIC AND
    (NOT YW1_BUILD_BOOTSTRAP OR NOT YW1_BOOTSTRAP_EXTERNAL_DIAGNOSTICS OR
     YW1_BOOTSTRAP_SD_LOGGING OR YW1_BOOTSTRAP_FS_DIAGNOSTIC))
    message(FATAL_ERROR "Exlaunch diagnostics require bootstrap/external diagnostics and disable FS profiles")
endif()
if(YW1_BOOTSTRAP_EXTERNAL_DIAGNOSTICS AND NOT YW1_BOOTSTRAP_FS_DIAGNOSTIC AND
    NOT YW1_BOOTSTRAP_EXLAUNCH_DIAGNOSTIC AND NOT YW1_BOOTSTRAP_EGL_DIAGNOSTIC)
    message(FATAL_ERROR "External diagnostics require the isolated FS diagnostic module")
endif()
if(YW1_BUILD_DIAGNOSTICS_COMPANION AND NOT YW1_BOOTSTRAP_EXTERNAL_DIAGNOSTICS)
    message(FATAL_ERROR "The companion requires the matching diagnostic module")
endif()
if(YW1_BOOTSTRAP_FS_DIAGNOSTIC AND NOT YW1_BOOTSTRAP_SD_LOGGING)
    message(FATAL_ERROR "FS diagnostics require YW1_BOOTSTRAP_SD_LOGGING")
endif()
if(YW1_BOOTSTRAP_SD_LOGGING AND NOT YW1_BUILD_BOOTSTRAP)
    message(FATAL_ERROR "SD logging requires YW1_BUILD_BOOTSTRAP")
endif()
if(YW1_BUILD_BOOTSTRAP)
    include(cmake/bootstrap_module.cmake)
elseif(YW1_COMPANION_EGL_GOT_PROBE)
    include(cmake/diagnostic_baseline.cmake)
endif()
if(YW1_BUILD_DIAGNOSTICS_COMPANION)
    include(cmake/diagnostics_companion.cmake)
endif()
if(YW1_BUILD_IMGUI_REFERENCE)
    include(cmake/imgui_reference.cmake)
endif()
if(YW1_BUILD_RENDER_FOUNDATION)
    include(cmake/render_foundation.cmake)
endif()
